14/// Result from a one-shot query to Claude.
15#[derive(Debug, Clone)]
16pub struct QueryResult {
17    /// The text content of the response.
18    pub content: String,
19    /// Token usage statistics, if available.
20    pub usage: Option<TokenUsage>,
21    /// The reason the response finished (e.g., "end_turn", "max_tokens").
22    pub finish_reason: String,
23}

25/// Perform a one-shot query to Claude Code.
26///
27/// This is a convenience function for simple, stateless interactions where you do not
28/// need bidirectional communication or conversation management. It creates a temporary
29/// client, sends a single prompt, and returns the complete response.
30///
31/// # Arguments
32///
33/// * `prompt` - The prompt to send to Claude.
34/// * `options` - Optional configuration options. If None, default options are used.
35///
36/// # Returns
37///
38/// A `QueryResult` containing the response content, usage statistics, and finish reason.
39///
40/// # Example
41///
42/// ```rust
43/// use claude_agent_sdk::query;
44///
45/// async fn example() -> Result<(), Box<dyn std::error::Error>> {
46///     let result = query("What is Rust?", None).await?;
47///     println!("{}", result.content);
48///     Ok(())
49/// }
50/// ```
51pub async fn query(
52    prompt: impl Into<String>,
53    options: Option<ClaudeAgentOptions>,
54) -> Result<QueryResult> {
55    let prompt = prompt.into();
56    let options = options.unwrap_or_default();
57
58    // Create transport with options
59    let transport_options = TransportOptions::from(&options);
60    let mut transport = SubprocessCLITransport::new(transport_options);
61
62    // Connect to the CLI
63    transport.connect().await?;
64
65    // Build and send the user message
66    let user_message = serde_json::json!({
67        "type": "user",
68        "message": {
69            "kind": "text",
70            "text": prompt
71        }
72    });
73
74    transport
75        .write(format!("{}\n", user_message).as_bytes())
76        .await?;
77
78    // Collect all text content from assistant messages
79    let mut content_parts: Vec<String> = Vec::new();
80    let mut usage: Option<TokenUsage> = None;
81    let mut finish_reason = String::from("unknown");
82
83    // Read messages until we get the result
84    loop {
85        match transport.read().await? {
86            Some(data) => {
87                let line = String::from_utf8_lossy(&data);
88                match serde_json::from_str::<Message>(&line) {
89                    Ok(Message::AssistantMsg { content, .. }) => {
90                        // Extract text content from assistant message
91                        for block in &content.content {
92                            if let crate::types::ContentBlock::Text { text } = block {
93                                content_parts.push(text.clone());
94                            }
95                        }
96                    }
97                    Ok(Message::ResultMsg {
98                        usage: msg_usage,
99                        stop_reason,
100                        result,
101                        ..
102                    }) => {
103                        // Capture final result data
104                        if let Some(result_text) = result {
105                            // If no content was collected from assistant messages,
106                            // use the result text
107                            if content_parts.is_empty() {
108                                content_parts.push(result_text);
109                            }
110                        }
111
112                        // Extract usage from the result message
113                        if let Some(u) = msg_usage {
114                            usage = extract_token_usage(&u);
115                        }
116
117                        // Extract finish reason
118                        if let Some(reason) = stop_reason {
119                            finish_reason = reason;
120                        }
121
122                        break;
123                    }
124                    Ok(_) => {
125                        // Ignore other message types
126                    }
127                    Err(e) => {
128                        return Err(ClaudeSDKError::Serialization(e));
129                    }
130                }
131            }
132            None => {
133                // End of stream
134                break;
135            }
136        }
137    }
138
139    // Close the transport
140    transport.close().await?;
141
142    let content = if content_parts.is_empty() {
143        String::new()
144    } else {
145        content_parts.join("")
146    };
147
148    Ok(QueryResult {
149        content,
150        usage,
151        finish_reason,
152    })
153}

155/// Extract TokenUsage from a JSON map.
156fn extract_token_usage(usage_map: &serde_json::Map<String, serde_json::Value>) -> Option<TokenUsage> {
157    let input_tokens = usage_map
158        .get("input_tokens")
159        .and_then(|v| v.as_i64())
160        .map(|v| v as i32)?;
161    let output_tokens = usage_map
162        .get("output_tokens")
163        .and_then(|v| v.as_i64())
164        .map(|v| v as i32)?;
165    let total_tokens = usage_map
166        .get("total_tokens")
167        .and_then(|v| v.as_i64())
168        .map(|v| v as i32)?;
169
170    Some(TokenUsage {
171        input_tokens,
172        output_tokens,
173        total_tokens,
174    })
175}
